programmers-42861 — 섬 연결하기

출처

프로그래머스 42861 — 섬 연결하기 (C++)

문제

원문 문제를 요약한 내용입니다. 전체 설명과 그림은 출처 링크에서 확인할 수 있습니다.

섬 n 개와 건설 가능한 다리의 비용이 주어집니다. 모든 섬 사이를 오갈 수 있도록 다리를 건설하는 데 필요한 최소 총비용을 반환합니다. 이동할 때 여러 다리를 거쳐도 됩니다.

함수와 제약 조건

  • 함수: int solution(int n, vector<vector<int>> costs)
  • n 은 1 이상 100 이하입니다.
  • costs 의 항목 수는 최대 n * (n - 1) / 2 개입니다.
  • 각 항목 [a, b, cost] 는 두 섬과 그 사이에 다리를 건설하는 비용을 나타냅니다.
  • 다리는 양방향으로 이동할 수 있습니다. 두 섬의 순서를 뒤집은 경우까지 포함해 같은 섬 쌍은 한 번만 주어집니다.
  • costs 에 없는 섬 쌍은 다리로 직접 연결할 수 없습니다.
  • 주어진 연결만으로 모든 섬을 연결할 수 있습니다.

예시

n costs 반환값
4 [[0,1,1],[0,2,2],[1,2,5],[1,3,1],[2,3,8]] 4

풀이

#include <string>
#include <vector>
#include <numeric>
#include <algorithm>

using namespace std;

class Edge {
    public: 
    int u;
    int v;
    int w;

    Edge(int u, int v, int w) : u(u), v(v), w(w) {}

    bool operator < (const Edge& other) {
        return this->w < other.w;
    }
};

class DS {
    private:
    vector<int> parent;
    vector<int> rank;

    public:
    DS(int size) {
        parent.resize(size+1);
        rank.resize(size+1);
        iota(parent.begin(), parent.end(), 0);
        rank.assign(size+1, 0);
    }

    int find(int x) {
        if (parent[x] == x) return x;
        return parent[x] = find(parent[x]);
    }

    bool unite(int x, int y) {
        int rootX = find(x);
        int rootY = find(y);

        if (rootX == rootY) return false;

        if (rank[rootX] < rank[rootY]) {
            parent[rootX] = rootY;
        }
        else if (rank[rootX] > rank[rootY]) {
            parent[rootY] = rootX;
        }
        else {
            parent[rootY] = rootX;
            rank[rootX] += 1;
        } 

        return true;
    }
};

int solution(int n, vector<vector<int>> costs) {
    int answer = 0;

    sort(costs.begin(), costs.end(), [](const vector<int>& a, const vector<int>& b) {
        return a[2] < b[2];
    });

    DS ds(n);
    int ec = 0;
    for (auto e : costs) {
        if (ds.unite(e[0], e[1]) == true) {
            answer += e[2];
            ec += 1;
        }

        if (ec == n - 1) break;
    }
    return answer;
}
